def _ensure_credentials_file(root: ctk.CTk) -> bool:
    """If the credentials file is missing, prompt the user to create it.
    Returns True if the file now exists (either already existed or was just created),
    False if the user cancelled.
    """
    if SETTINGS.is_file():
        return True

    # First-run dialog
    dialog = ctk.CTkToplevel(root)
    dialog.title("首次运行 - 配置微信读书凭证")
    dialog.geometry("520x380")
    dialog.resizable(False, False)
    dialog.transient(root)
    dialog.grab_set()
    dialog.protocol("WM_DELETE_WINDOW", lambda: None)  # prevent closing

    # Center the dialog
    dialog.update_idletasks()
    x = root.winfo_rootx() + (root.winfo_width() - 520) // 2
    y = root.winfo_rooty() + (root.winfo_height() - 380) // 2
    dialog.geometry(f"520x380+{x}+{y}")

    ctk.CTkLabel(
        dialog,
        text="首次运行：配置微信读书凭证",
        font=_font(14, "bold"),
        text_color="white",
    ).pack(pady=(20, 8))

    ctk.CTkLabel(
        dialog,
        text=(
            "程序需要两个微信读书的凭证才能工作：\n\n"
            "1. api_key (wrk-...) 和 user_vid\n"
            "   -> 浏览器打开 https://weread.qq.com/r/weread-skills 登录后复制\n\n"
            "2. 扫码登录后会自动获取 wr_skey (cookie)\n"
            "   这一步在程序里点"扫码登录"完成"
        ),
        justify="left",
        wraplength=460,
        justify="left",
    ).pack(padx=20, pady=(0, 16), anchor="w")

    # api_key input
    ctk.CTkLabel(dialog, text="api_key (wrk-...)", font=_font(12)).pack(anchor="w", padx=20)
    api_key_var = ctk.StringVar()
    api_key_entry = ctk.CTkEntry(dialog, textvariable=api_key_var, width=440, height=36, placeholder_text="wrk-xxxxxxxxxxxxxxxx")
    api_key_entry.pack(padx=20, pady=(0, 12), fill="x")

    # user_vid input
    ctk.CTkLabel(dialog, text="user_vid (纯数字)", font=_font(12)).pack(anchor="w", padx=20, pady=(12, 0))
    user_vid_var = ctk.StringVar()
    user_vid_entry = ctk.CTkEntry(dialog, textvariable=user_vid_var, width=440, height=36, placeholder_text="纯数字，例如 123456789")
    user_vid_entry.pack(padx=20, pady=(0, 20), fill="x")

    # Link to the skills page
    link_label = ctk.CTkLabel(
        dialog,
        text="去网页版获取凭证 -> weread.qq.com/r/weread-skills",
        text_color="#4A9EFF",
        cursor="hand2",
        font=_font(11, "underline"),
    )
    link_label.pack(anchor="w", padx=20)
    link_label.bind("<Button-1>", lambda e: webbrowser.open("https://weread.qq.com/r/weread-skills"))

    result = {"ok": False}

    def on_ok():
        api_key = api_key_var.get().strip()
        user_vid = user_vid_var.get().strip()
        if not api_key or not user_vid:
            tk_messagebox.showwarning("缺少信息", "请填写 api_key 和 user_vid")
            return
        if not api_key.startswith("wrk-"):
            tk_messagebox.showwarning("格式错误", "api_key 应以 wrk- 开头")
            return
        if not user_vid.isdigit():
            tk_messagebox.showwarning("格式错误", "user_vid 应为纯数字")
            return

        # Write the lua file
        lua_content = f'''-- ./settings/weread.lua
return {{
  ["account"] = {{
    ["login_method"] = "qr",
    ["login_time"] = {int(time.time())},
    ["name"] = "",
    ["user_vid"] = "{user_vid}",
  }},
  ["api_key"] = "{api_key}",
  ["auth_schema_version"] = 1,
  ["books"] = {{}},
}}
'''
        try:
            SETTINGS.parent.mkdir(parents=True, exist_ok=True)
            SETTINGS.write_text(lua_content, encoding="utf-8")
            dialog.destroy()
            dialog._result = True
        except Exception as e:
            tk_messagebox.showerror("写入失败", f"无法写入凭证文件：{e}")

    def on_ok():
        api_key = api_key_var.get().strip()
        user_vid = user_vid_var.get().strip()
        if not api_key or not user_vid:
            tk_messagebox.showwarning("缺少信息", "请填写 api_key 和 user_vid")
            return
        if not api_key.startswith("wrk-"):
            tk_messagebox.showwarning("格式错误", "api_key 应以 wrk- 开头")
            return
        if not user_vid.isdigit():
            tk_messagebox.showwarning("格式错误", "user_vid 应为纯数字")
            return

        # Write the lua file
        lua_content = f'''-- ./settings/weread.lua
return {{
  ["account"] = {{
    ["login_method"] = "qr",
    ["login_time"] = {int(time.time())},
    ["name"] = "",
    ["user_vid"] = "{user_vid}",
  }},
  ["api_key"] = "{api_key}",
  ["auth_schema_version"] = 1,
  ["books"] = {{}},
}}
'''
        try:
            SETTINGS.parent.mkdir(parents=True, exist_ok=True)
            SETTINGS.write_text(lua_content, encoding="utf-8")
            dialog.destroy()
            dialog._result = True
        except Exception as e:
            tk_messagebox.showerror("写入失败", f"无法写入凭证文件：{e}")

    def on_cancel():
        dialog.destroy()

    ctk.CTkButton(dialog, text="保存并继续", command=on_ok, width=140, height=36).pack(pady=(8, 20))
    ctk.CTkButton(dialog, text="取消", command=on_cancel, width=100, height=32, fg_color="transparent", border_width=1).pack(pady=(0, 16))

    dialog.wait_window()
    return getattr(dialog, "_result", False)
